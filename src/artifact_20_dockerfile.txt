FROM node:20-alpine AS builder

WORKDIR /usr/src/app

COPY client/package*.json ./
RUN npm ci

COPY client/ ./
RUN npm run build

FROM node:20-alpine AS runner

WORKDIR /app
ENV NODE_ENV=production

COPY client/package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

USER node

EXPOSE 8080
CMD ["node", "dist/daemon.js"]
